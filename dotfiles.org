  # Prepend DIR to PATH if it exists and isn't already there; the session
  # (uwsm) may already have imported PATH from an earlier login shell
  path_prepend() {
      [ -d "$1" ] || return
      case ":$PATH:" in
          *":$1:"*) ;;
          *) PATH="$1:$PATH" ;;
      esac
  }
  path_prepend "$HOME/.local/bin"

  unset -f path_prepend
  [ -f "$HOME/.cargo/env" ] && . "$HOME/.cargo/env"

      . "$HOME/.bashrc"

  shopt -s globstar     # matches '**' recursively in all sub-dirs (**/*.sh)
  shopt -s histappend   # append to the history file, don't overwrite it
  shopt -s lithist      # conserve newlines and formatting

  # ==============================================================================
  # HISTORY SETTINGS
  # ==============================================================================
  export HISTSIZE=250000
  export HISTFILESIZE=250000
  export HISTCONTROL=ignoreboth
  export HISTIGNORE="exit:pwd:clear"
  export HISTTIMEFORMAT="%F %T "

  # Initialize PROMPT_COMMAND with history sync
  PROMPT_COMMAND="${PROMPT_COMMAND:+$PROMPT_COMMAND; }history -a"

  # ==============================================================================
  # PROMPT SETTINGS
  # ==============================================================================
  export PROMPT_DIRTRIM=2
  export PS1="[\h][\w]\$ "

  # ~/.bash_ps1.sh is generated by matugen; re-source it only when it is
  # newer than this shell's stamp file (builtin test and redirect, no forks)
  _ps1_stamp="${XDG_RUNTIME_DIR:-/tmp}/bash-ps1.$$"
  _ps1_refresh() {
      if [[ ~/.bash_ps1.sh -nt $_ps1_stamp ]]; then
          source ~/.bash_ps1.sh
          : > "$_ps1_stamp"
      fi
  }

  if [[ "$TERM" != "dumb" ]]; then
      _ps1_refresh
      PROMPT_COMMAND="${PROMPT_COMMAND:+$PROMPT_COMMAND; }_ps1_refresh"
  fi

  # ==============================================================================
  # ALIASES & COLORS
  # ==============================================================================
  if [ -x /usr/bin/dircolors ]; then
      test -r ~/.dircolors && eval "$(dircolors -b ~/.dircolors)" || eval "$(dircolors -b)"
      alias ls='ls --color=auto'
      alias grep='grep --color=auto'
  fi

  alias ll='ls -alF'
  alias la='ls -A'
  alias l='ls -CF'

  # Load external aliases if they exist
  [ -f "$HOME/.bash_aliases" ] && source "$HOME/.bash_aliases"

  # ==============================================================================
  # EXTERNAL TOOLS & COMPLETIONS
  # ==============================================================================

  # Bash Completion
  if ! shopt -oq posix; then
      if [ -f /usr/share/bash-completion/bash_completion ]; then
          . /usr/share/bash-completion/bash_completion
      elif [ -f /etc/bash_completion ]; then
          . /etc/bash_completion
      fi
  fi

  # FZF (Key bindings and Completion)
  if [ -f /usr/share/fzf/key-bindings.bash ]; then
      source /usr/share/fzf/key-bindings.bash
      source /usr/share/fzf/completion.bash

      export FZF_DEFAULT_OPTS='--height=100% --layout=reverse --border'

  # zoxide (smart cd)
  if command -v zoxide >/dev/null 2>&1; then
      export _ZO_ECHO=1
      eval "$(zoxide init bash)"

  # keychain (ssh agent)
  # keychain itself takes ~300ms, which every shell (and Emacs's
  # exec-path-from-shell) pays. Reuse the agent from its env file and only
  # run keychain when the agent is gone or has no keys loaded.
  # Only ask for the passphrase in interactive shells: the display manager
  # runs a login shell with no terminal, and keychain 3 would block session
  # startup waiting for Enter.
  if command -v keychain >/dev/null 2>&1 && [ -f ~/.ssh/id_ed25519 ]; then
      [ -f ~/.keychain/"$HOSTNAME"-sh ] && . ~/.keychain/"$HOSTNAME"-sh
      if ! ssh-add -l >/dev/null 2>&1; then
          if [[ $- == *i* ]]; then
              eval "$(keychain --eval --nogui --quiet ~/.ssh/id_ed25519)"
          else
              eval "$(keychain --eval --nogui --quiet --noask)"
          fi
      fi
  fi

  if [[ "$INSIDE_EMACS" = 'ghostel' ]]; then
      # Open a file in Emacs from the terminal
      e()   { ghostel_cmd find-file-other-window "$@"; }

      # Open dired in another window
      dow() { ghostel_cmd dired-other-window "$@"; }

      # Open magit for the current directory
      gst() { ghostel_cmd magit-status-setup-buffer "$(pwd)"; }
  fi

  # Machine-specific overrides
  [ -f ~/.bashrc.local ] && . ~/.bashrc.local

Colours for the [[https://github.com/uiriansan/SilentSDDM][SilentSDDM]] login screen. The greeter runs as the =sddm= user and cannot read =$HOME=, so matugen writes to a directory we own outside it, and the theme reads it through symlinks. SDDM layers =<ConfigFile>.user= over the theme's own config, and Silent looks up backgrounds in its =backgrounds/= directory, hence the two links. =20-theme.conf= sorts after EndeavourOS's =10-endeavouros.conf=, which selects =eos-breeze=.
#+begin_src emacs-lisp

     (:unless (file-symlink-p "/usr/share/sddm/themes/silent/configs/default.conf.user")
      :run "sudo ln -s /var/lib/greeter-theme/default.conf.user /usr/share/sddm/themes/silent/configs/default.conf.user")
     (:unless (file-symlink-p "/usr/share/sddm/themes/silent/backgrounds/matugen.png")
      :run "sudo ln -s /var/lib/greeter-theme/wallpaper /usr/share/sddm/themes/silent/backgrounds/matugen.png")
     (:unless (file-exists-p "/etc/sddm.conf.d/20-theme.conf")

We only want to set =post_command= in waypaper's config and leave the rest (which waypaper writes itself) untouched. The file is created if missing and only written when the line changes, so this is cheap to run on every save.
#+begin_src emacs-lisp
  (defun hesam-waypaper--inject-postcommand ()
    "Set post_command in waypaper's config.ini, creating the file if needed.
  The file is only written when its contents change."
    (let ((path (expand-file-name "waypaper/config.ini" (xdg-config-home)))
          (line "post_command = emacsclient --eval \"(hesam-matugen-apply nil \\\"$wallpaper\\\")\""))
      (with-temp-buffer
        (if (file-exists-p path)
            (insert-file-contents path)
          (insert "[Settings]\n"))
        (let ((old (buffer-string)))
          (goto-char (point-min))
          (if (re-search-forward "^post_command =.*$" nil t)
              (replace-match line t t)
            (goto-char (point-max))
            (unless (bolp) (insert "\n"))
            (insert line "\n"))
          (unless (string= old (buffer-string))
            (make-directory (file-name-directory path) t)
            (write-region nil nil path))))))

  (hesam-waypaper--inject-postcommand)
#+end_src

** check update
The script uses =checkupdates= from =pacman-contrib=; the timer runs it weekly.
#+begin_src emacs-lisp
  (system-pkg pacman-contrib
    :exe checkupdates
    :setup
    ((:unless (system-pkg-unit-enabled-p "check-updates.timer")
      :run "systemctl --user enable --now check-updates.timer")))
#+end_src
*** script
#+begin_src bash :tangle ~/.local/bin/check-updates.sh :shebang "#!/usr/bin/bash"
  # checkupdates exits 0 with updates, 2 without and anything else on
  # failure (e.g. no network right after resume), so retry a few times.
  for attempt in 1 2 3 4; do
      official=$(checkupdates 2>/dev/null)
      status=$?
      [ "$status" -eq 0 ] || [ "$status" -eq 2 ] && break
      [ "$attempt" -lt 4 ] && sleep 10m
  done

  if [ "$status" -ne 0 ] && [ "$status" -ne 2 ]; then
      notify-send -u critical "System Update" "Update check failed. See systemctl --user status check-updates."
      exit 1
  fi

  official_count=$(grep -c . <<< "$official")
  aur_count=$(yay -Qua 2>/dev/null | wc -l)
  total_count=$((official_count + aur_count))

  if [ "$total_count" -gt 0 ]; then
      notify-send -u critical "System Update" "$total_count packages are available to update."
  fi
#+end_src

** on battery
Succeeds when a system battery is present and no charger is connected. Desktops count as on AC. Peripheral batteries (mice, headsets) report =scope= =Device= and are skipped.
#+begin_src bash :tangle ~/.local/bin/on-battery.sh :shebang "#!/bin/sh"
  battery=no
  for s in /sys/class/power_supply/*; do
      [ "$(cat "$s/scope" 2>/dev/null)" = Device ] && continue
      case $(cat "$s/type") in
          Mains|USB) [ "$(cat "$s/online" 2>/dev/null)" = 1 ] && exit 1 ;;
          Battery) battery=yes ;;
      esac
  done
  [ "$battery" = yes ]
#+end_src
** external backlight
Control backlight of external monitors with =ddcutil=.
#+begin_src bash :tangle ~/.local/bin/external-backlight.sh :shebang "#!/bin/bash"
  # setvcp arguments are + N to increase N or - N to decrease
  # $* joins all arguments (e.g. + and 10) into a single string
  ddcutil setvcp 10 $* --noverify

  # Get the new value to show in the OSD
  # --terse format: VCP 10 C 50 100 (where 50 is current, 100 is max)
  current=$(ddcutil getvcp 10 --terse | awk '{print $4}')

  # Trigger SwayOSD to show the bar
  swayosd-client --brightness "$current"
#+end_src
